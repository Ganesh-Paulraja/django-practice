------------------dynamic values from view.py to home.html---------------------------
views.py -->
def home(request):
    title = "Jclick Solution"
    course = ['php', 'java', 'dotnet', 'python']
    detail = {'name': 'Ajin', 'address': 'IN'}
    return render(request, 'home.html', {'title': title, 'courses': course, 'details': detail})
